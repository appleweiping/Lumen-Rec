\section{Setting and Notation}
\label{sec:prelim}

\paragraph{Yes/no recommender and token confidence.}
A pointwise yes/no recommender~\citep{bao2023tallrec,zhang2025collm} scores a user $u$ and a candidate item $i$ by asking a
binary question $q$ about the pair (default $q=\textsf{like}$, ``Would this user like the candidate item?'') and reading the
next-token distribution of a language model $f$, zero-shot or with a LoRA adapter fitted to yes/no labels. Its
\emph{confidence} is the log-odds
\begin{equation}\label{eq:logit}
  \ell(u,i)=\log P_f(\texttt{Yes}\mid x_q(u,i))-\log P_f(\texttt{No}\mid x_q(u,i)),
\end{equation}
where $P_f(\texttt{Yes}\mid x)$ sums the probabilities of the vocabulary ids that decode, stripped and lower-cased, to ``yes''
(likewise \texttt{No}); candidates are ranked by $\ell$.

\paragraph{Panels and metrics.}
A \emph{unit} is a user (rated panels) or a prediction event (next-item panels) with candidate set $C_u$ and labels
$y(u,i)\in\{0,1\}$. \emph{Rated panels} show the user's earlier ratings as history and use the user's own later rated items as
candidates, labelled by whether the user liked them (Section~\ref{sec:protocol}). With $c(x)=1,\tfrac12,0$ for $x>0,\,x=0,\,x<0$,
the per-user AUC of a score $s$ and its mean over users with both classes (UAUC~\citep{zhang2025collm,zhang2024binllm}) are
\[
  \mathrm{AUC}_u(s)=\frac{1}{|C_u^+||C_u^-|}\sum_{i\in C_u^+}\sum_{j\in C_u^-}c(s_i-s_j),\quad
  \mathrm{UAUC}=\operatorname*{mean}_{u}\mathrm{AUC}_u .
\]
\emph{Same-candidate next-item panels} hold, per event, one positive and 100 popularity-sampled negatives, ranked identically by
every system (not full-catalogue ranking~\citep{krichene2020sampled}). With $a$ candidates above the positive and $t$ tied with it,
its rank $R$ is uniform on $1{+}a,\dots,1{+}a{+}t$, and HR@$k$, NDCG@$k$ and MRR are the expectations of $\mathbf 1[R{\le}k]$,
$\mathbf 1[R{\le}k]/\log_2(1{+}R)$ and $1/R$ ($k{=}10$).

\paragraph{Decomposition and audit quantities.}
Let $\pi(i)=\mathbb E_D[\ell(D,i)]$ be the \emph{item prior}, the expected confidence for $i$ under the history of a donor
$D$ who has not seen $i$ (estimated by the mean $\hat\pi(i)$ over $K=8$ donors). With the user offset
$b_u=|C_u|^{-1}\sum_{i\in C_u}[\ell(u,i)-\pi(i)]$, the remainder is the \emph{pair-specific residual} $e$:
\begin{equation}\label{eq:decomp}
  \ell(u,i)=b_u+\pi(i)+e(u,i),\qquad \textstyle\sum_{i\in C_u}e(u,i)=0,
\end{equation}
so within a unit only $\pi+e$ orders candidates. The residual $e$ is not personal evidence by construction: it also holds
donor-population mismatch (donors have not consumed $i$), user-by-prior interaction and history noise unrelated to the label. The
audit therefore measures personal information by prediction. On rated panels it reports (estimators: Section~\ref{sec:tools}) the
\emph{information gain} $\mathcal G$, the UAUC gain of cross-fitted stackers when $\hat e=\ell-\hat\pi$ is added to $\hat\pi$ and
the prior-only item mean $m(i)$ (other users' mean rating of $i$ before the candidate's timestamp), beside the analogous gain
$\mathcal G_{\rm CF}$ of a matrix-factorisation residual over $m$; the \emph{non-prior share} $1-\rho^2/r_8$, the share of
the within-unit variance of $\ell$ that $\hat\pi$ does not explain linearly ($\rho$ the pooled within-unit correlation of $\ell$
and $\hat\pi$, $r_8$ the Spearman--Brown reliability~\citep{spearman1910correlation,brown1910experimental} of the 8-donor mean),
with the variance share of $e$ itself beside it; and the \emph{popularity link}, the partial Spearman correlation of $\ell$ and
of $\hat\pi$ with log popularity given $m(i)$.

\paragraph{Where can confidence change a ranking?}
\begin{lemma}[Invariance]\label{lem:invariance}\label{prop:monotone}
If every $g_u:\mathbb R\to\mathbb R$ is strictly increasing and $s'_i=g_u(s_i)$ for $i\in C_u$, then
\[
  \operatorname{sgn}(s'_i-s'_j)=\operatorname{sgn}(s_i-s_j)\qquad\text{for all } i,j\in C_u,
\]
so $\mathrm{AUC}_u$, HR@$k$, NDCG@$k$ and MRR, which depend on $s$ only through these signs (via $a$, $t$ and $c$), are
unchanged under either tie convention.
\end{lemma}
\noindent Strict monotonicity preserves $>$, $=$ and $<$. Temperature~\citep{guo2017calibration} and Platt
scaling~\citep{platt2000probabilities} with positive slope, removal of $b_u$ and the list softmax $w_i\propto\exp(\ell_i/T)$ are such
maps: per-user calibration cannot change UAUC or within-unit NDCG. Merely non-decreasing maps (isotonic
regression~\citep{zadrozny2002transforming}) can, through ties.
\begin{proposition}[Item-dependent corrections]\label{prop:itemdep}
For continuous, strictly increasing $g_{u,i}$ and $s'_i=g_{u,i}(s_i)$, the weak order of $C_u$ is unchanged for every score
vector if and only if all $g_{u,i}$ coincide; for $s'=s-\lambda b(i)$, a pair with $s_i>s_j$ is exchanged or tied exactly when
\[
  s_i-s_j\le\lambda\,\bigl(b(i)-b(j)\bigr).
\]
\end{proposition}
\noindent ``If'' is Lemma~\ref{lem:invariance}; if $g_{u,i}(x)<g_{u,j}(x)$, continuity gives $\varepsilon>0$ with
$g_{u,i}(x{+}\varepsilon)<g_{u,j}(x)$, so $s_i=x{+}\varepsilon>s_j=x$ is reversed. Contextual~\citep{zhao2021calibrate} and batch
calibration~\citep{zhou2024batch} subtract a prior that, for a binary question, is a constant of the unit; candidate-conditioned priors
(the no-history prompt, in the spirit of PMI scoring~\citep{holtzman2021surface}; the swap prior $\hat\pi$) and second readouts of the
pair (a mirrored or paraphrased question) are item-dependent, and as reordering is not improving each is compared with an
equal-compute control. Across units the lemma is silent: offsets change pooled AUC, thresholds and selective serving, and training
changes $f$.
